gateway.yaml
name: events_gateway

listen:
  addr: :8080
  auth:
    type: bearer
    keys_from: BREVIS_INGEST_KEYS

streams:
  - name: clicks
    path: /v1/clicks
    format: json
    identity:
      provider: web
      entity: click
      source_key: event_id
      record_ts: occurred_at
    sink:
      type: files
      path: /var/landed/
    dead_letter:
      type: files
      path: /var/landed/dead-letter/
